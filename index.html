function updateArchivoDashboard(data) { // Conteo real y limpio de solicitudes únicas basadas en la Columna A let solicitudesUnicas = new Set(); let totalDocs = 0; let totalImagenes = 0; let sumMetricaTiempo = 0; let countTiempos = 0; let condicionesMap = {}; let oficinasMap = {}; data.forEach(r => { // Obtenemos el valor de la columna A y lo limpiamos de espacios const valSolicitudId = getFieldValue(r, ['N°', 'Nº', 'N° SOLICITUD', 'SOLICITUD', 'ID']); const valDocsStr = getFieldValue(r, ['Nª DE DOC', 'N° DE DOC', 'CANTIDAD', 'DOCUMENTOS']); const valImgsStr = getFieldValue(r, ['IMAGENES', 'IMÁGENES', 'FOLIOS']); const valMetricaStr = getFieldValue(r, ['METRICA', 'MÉTRICA', 'NIVEL', 'TIEMPO']); const condicion = getFieldValue(r, ['CONDICIÓN DE SERVICIO', 'CONDICION DE SERVICIO', 'CONDICION']).toUpperCase() || 'OTROS'; const oficina = getFieldValue(r, ['OFICINA', 'UNIDAD ORGÁNICA']).toUpperCase() || 'SIN OFICINA'; if (valSolicitudId !== '') { // Normalizamos a string limpio para que las repeticiones (como el 158) cuenten como uno solo solicitudesUnicas.add(String(valSolicitudId).trim()); } let valDocs = parseFloat(String(valDocsStr).replace(',', '.')); if (isNaN(valDocs) || valDocs < 0) valDocs = 0; let valImgs = parseFloat(String(valImgsStr).replace(',', '.')); if (isNaN(valImgs) || valImgs < 0) valImgs = 0; let valMetrica = parseFloat(String(valMetricaStr).replace(',', '.')); if (!isNaN(valMetrica) && valMetrica > 0) { sumMetricaTiempo += valMetrica; countTiempos++; } totalDocs += valDocs; totalImagenes += valImgs; condicionesMap[condicion] = (condicionesMap[condicion] || 0) + 1; if (!oficinasMap[oficina]) { oficinasMap[oficina] = { servicios: 0, docs: 0, imgs: 0 }; } oficinasMap[oficina].servicios += 1; oficinasMap[oficina].docs += valDocs; oficinasMap[oficina].imgs += valImgs; }); // Si el Set detecta las solicitudes únicas correctamente, usará el tamaño real. // Si por alguna razón la columna A viniera vacía, se protege con el total de filas. const totalServicios = solicitudesUnicas.size > 0 ? solicitudesUnicas.size : data.length; const promDocsPorServicio = totalServicios > 0 ? (totalDocs / totalServicios).toFixed(1) : '0.0'; const promImgsPorDoc = totalDocs > 0 ? (totalImagenes / totalDocs).toFixed(1) : '0.0'; const promTiempoPorServicio = countTiempos > 0 ? (sumMetricaTiempo / countTiempos).toFixed(1) : '0.0'; const promTiempoPorDoc = totalDocs > 0 ? (sumMetricaTiempo / totalDocs).toFixed(2) : '0.00'; document.getElementById('kpi-total-servicios').innerText = totalServicios; document.getElementById('kpi-total-docs-archivo').innerText = totalDocs.toLocaleString(); document.getElementById('kpi-total-imagenes').innerText = totalImagenes.toLocaleString(); document.getElementById('kpi-prom-docs-servicio').innerText = `${promDocsPorServicio} docs / servicio`; document.getElementById('kpi-prom-imgs-doc').innerText = `${promImgsPorDoc} imágenes / doc`; document.getElementById('kpi-prom-tiempo-servicio').innerText = promTiempoPorServicio; document.getElementById('kpi-prom-tiempo-doc').innerText = promTiempoPorDoc; renderChartsArchivo(condicionesMap, oficinasMap); renderTableOficinasArchivo(oficinasMap, totalServicios); }
